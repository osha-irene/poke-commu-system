이미지 저장 위치: 모든 이미지는 public/ 폴더 안에 넣으면 됩니다.
예: public/img/backgrounds/app-bg.png → 경로: /img/backgrounds/app-bg.png

예시 1 — 앱 전체 배경 이미지 적용

// src/styles/layout.js export const APP_SHELL = { root: 'h-screen flex', // bg-gray-50 제거 backgroundImage: '/img/backgrounds/app-bg.png', backgroundSize: 'cover', backgroundPosition: 'center', backgroundAttachment: 'fixed', // 스크롤해도 배경 고정 overlayColor: 'rgba(0, 0, 0, 0.35)', // 어두운 막 (가독성) };

예시 2 — 로그인 화면 배경 + 카드 배경 적용

export const LOGIN = { root: 'min-h-screen flex items-center justify-center p-4', backgroundImage: '/img/backgrounds/login-bg.png', backgroundSize: 'cover', backgroundPosition:'center', overlayColor: 'rgba(0, 0, 0, 0.5)', card: 'rounded-2xl shadow-xl p-8 w-full max-w-md', // bg-white 제거 cardBackgroundImage: '/img/backgrounds/login-card-bg.png', cardBackgroundSize: 'cover', cardOverlayColor: 'rgba(255, 255, 255, 0.88)', // 밝은 막 (폼 가독성) titleImage: '/img/login/title.png', // 텍스트 대신 이미지 타이틀 };

예시 3 — 사이드바 배경만 교체

export const SIDEBAR = { root: 'w-64 flex flex-col flex-shrink-0', // bg-white border-r 제거 backgroundImage: '/img/backgrounds/sidebar-bg.png', backgroundSize: 'cover', backgroundPosition:'center', overlayColor: 'rgba(0, 0, 0, 0.45)', // 텍스트 보이게 logo: { image: '/img/sidebar/logo.png', // 텍스트 로고 → 이미지로 }, };

'rgba(0,0,0,0.60)'
